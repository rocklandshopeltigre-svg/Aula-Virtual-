            const errorMsg = validatorFn(inputEl.value);

            if (errorMsg) {
                groupEl.classList.remove('valid');
                groupEl.classList.add('invalid');
                if (errorEl) errorEl.textContent = errorMsg;
                return false;
            } else {
                groupEl.classList.remove('invalid');
                groupEl.classList.add('valid');
                if (errorEl) errorEl.textContent = '';
                return true;
            }
        }

        /**
         * Configuración de validación en tiempo real en un formulario
         */
        function setupFormValidation(formEl, rules) {
            // Escuchar eventos input y blur
            Object.keys(rules).forEach(fieldId => {
                const inputEl = formEl.querySelector(`#${fieldId}`);
                if (!inputEl) return;

                const validatorFn = rules[fieldId];

                inputEl.addEventListener('input', () => validateField(inputEl, validatorFn));
                inputEl.addEventListener('blur', () => validateField(inputEl, validatorFn));
            });

            // Función evaluadora global del formulario
            return function isFormValid() {
                let isValid = true;
                Object.keys(rules).forEach(fieldId => {
                    const inputEl = formEl.querySelector(`#${fieldId}`);
                    if (inputEl) {
                        const fieldValid = validateField(inputEl, rules[fieldId]);
                        if (!fieldValid) isValid = false;
                    }
                });
                return isValid;
            };
        }

        // Renderizar Catálogo de Repuestos
        function renderCatalog() {

            const emptyState = document.getElementById('emptyCatalogState');

            const filteredParts = PARTS_DATABASE.filter(part => {
                const matchBrand = state.filterBrand === "ALL" || part.brand === state.filterBrand;
                const matchCategory = state.filterCategory === "ALL" || part.category === state.filterCategory;
                const matchStock = state.filterStock === "ALL" || part.stock === state.filterStock;
                
                const q = state.searchQuery.toLowerCase().trim();
                const matchQuery = !q || 
                    part.partNumber.toLowerCase().includes(q) ||
                    part.name.toLowerCase().includes(q) ||
                    part.brand.toLowerCase().includes(q) ||
                    part.engineCompatibility.toLowerCase().includes(q);

                return matchBrand && matchCategory && matchStock && matchQuery;
            });

            document.getElementById('catalogCount').textContent = filteredParts.length;

            if (filteredParts.length === 0) {
                grid.innerHTML = "";
                emptyState.style.display = "block";
                return;
            }

            emptyState.style.display = "none";
            grid.innerHTML = filteredParts.map(part => `
                <div class="part-card">
                    <div>
                        <div class="part-card-img-wrapper">
                            <img src="${part.image}" alt="${part.name}" class="part-card-img" onerror="this.src='https://placehold.co/600x400/1e293b/ff9900?text=${part.partNumber}'">

        /* ==========================================================================
           GESTIÓN DEL CARRITO DE COTIZACIÓN (RFQ)
           ========================================================================== */
        function addToCart(partId) {
            const part = PARTS_DATABASE.find(p => p.id === partId);
            if (!part) return;

            const existing = state.cart.find(item => item.id === partId);
            if (existing) {
                existing.qty += 1;
            } else {
                state.cart.push({ ...part, qty: 1 });
            }

            updateCartUI();
            showToast(`Añadido: Repuesto ${part.partNumber}`);
        }

        function updateCartQty(partId, delta) {
            const item = state.cart.find(i => i.id === partId);
            if (!item) return;

            item.qty += delta;
            if (item.qty <= 0) {
                state.cart = state.cart.filter(i => i.id !== partId);
            }
            updateCartUI();
        }

        function updateCartUI() {
            const badge = document.getElementById('cartBadgeCount');
            const totalQty = state.cart.reduce((sum, item) => sum + item.qty, 0);
            badge.textContent = totalQty;

            const listContainer = document.getElementById('cartItemsList');
            const totalDisplay = document.getElementById('rfqTotalDisplay');

            if (state.cart.length === 0) {
                listContainer.innerHTML = `
                    <div style="text-align: center; padding: 3rem 1rem; color: var(--text-muted);">
                        <div style="font-size: 2rem; margin-bottom: 0.5rem;">📋</div>
                        <p style="font-size: 0.8rem;">La lista de cotización está vacía.</p>
                    </div>
                `;
                totalDisplay.textContent = "$0.00 USD";
                return;
            }

            let grandTotal = 0;
            listContainer.innerHTML = state.cart.map(item => {
                const subtotal = item.price * item.qty;
                grandTotal += subtotal;
                return `
                    <div class="cart-item">
                        <div>
                            <div class="cart-item-part">${item.brand} | ${item.partNumber}</div>
                            <div class="cart-item-title">${item.name}</div>
                            <div style="font-size: 0.75rem; font-family: var(--font-mono); color: var(--text-muted);">$${item.price.toFixed(2)} c/u</div>
                        </div>
                        <div class="qty-controls">
                            <button type="button" class="qty-btn" onclick="updateCartQty('${item.id}', -1)">-</button>
                            <span style="font-size: 0.8rem; font-family: var(--font-mono); font-weight: 700; px-1;">${item.qty}</span>
                            <button type="button" class="qty-btn" onclick="updateCartQty('${item.id}', 1)">+</button>
                        </div>
                    </div>
                `;
            }).join('');

            totalDisplay.textContent = `$${grandTotal.toFixed(2)} USD`;
        }

        // Acciones del Cotizador con Validación Obligatoria del Formulario de Cliente
        function executeRFQAction(actionType) {
            if (state.cart.length === 0) {
                showToast("Agregue al menos un repuesto a la cotización.");
                return;
            }

            // Forzar validación de datos de contacto del cliente
            const isCustomerValid = validateRFQForm();
            if (!isCustomerValid) {
                showToast("Por favor complete los datos del solicitante en el formulario inferior.");
                return;
            }

            const clientName = document.getElementById('rfqName').value.trim();
            const clientEmail = document.getElementById('rfqEmail').value.trim();
            const clientPhone = document.getElementById('rfqPhone').value.trim();

            if (actionType === 'SUBMIT' || actionType === 'TXT') {
                let summary = "========================================================\n";
                summary += "       TMyC DE VENEZUELA C.A. - SOLICITUD DE COTIZACIÓN\n";
                summary += "========================================================\n\n";
                summary += `Cliente: ${clientName}\n`;
                summary += `Correo: ${clientEmail}\n`;
                summary += `Teléfono: ${clientPhone}\n`;
                summary += "Fecha: " + new Date().toLocaleString('es-VE') + "\n";
                summary += "--------------------------------------------------------\n\n";

                let total = 0;
                state.cart.forEach(item => {
                    const subtotal = item.price * item.qty;
                    total += subtotal;
                    summary += `• Marca: ${item.brand}\n  N° Parte: ${item.partNumber}\n  Item: ${item.name}\n  Cant: ${item.qty} x $${item.price.toFixed(2)} = $${subtotal.toFixed(2)} USD\n\n`;
                });

                summary += "--------------------------------------------------------\n";
                summary += `TOTAL ESTIMADO REFERENCIAL: $${total.toFixed(2)} USD\n`;

                if (actionType === 'TXT') {
                    const blob = new Blob([summary], { type: "text/plain;charset=utf-8" });
                    const link = document.createElement("a");
                    link.href = URL.createObjectURL(blob);
                    link.download = `Cotizacion_TMyC_${clientName.replace(/\s+/g, '_')}.txt`;
                    link.click();
                    showToast("Resumen de cotización descargado.");
                } else {
                    showToast("¡Solicitud enviada a ventas@tmycvenezuela.com!");
                    closeCartDrawer();
                }
            } else if (actionType === 'WHATSAPP') {
                let text = `Hola TMyC de Venezuela C.A., soy *${clientName}* (${clientEmail}). Deseo solicitar cotización oficial para:\n\n`;
                state.cart.forEach(item => {
                    text += `• ${item.brand} N° ${item.partNumber} (${item.name}) - Cant: ${item.qty}\n`;
                });
                const url = `https://wa.me/584145550199?text=${encodeURIComponent(text)}`;
                window.open(url, '_blank');
            }
        }
